class Solution:
    def isValid(self, s: str) -> bool:
        closers = {")": "(", "]": "[", "}": "{"}
        stack = []
        for ch in s:
            if ch in closers:
                if not stack or stack.pop() != closers[ch]:
                    return False
            else:
                stack.append(ch)
        return not stack
